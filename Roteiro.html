        // Tabs

        function openTab(cityName) {

            const contents = document.querySelectorAll('.tab-content');

            const buttons = document.querySelectorAll('.tab-button');


            contents.forEach(content => content.classList.remove('active'));

            buttons.forEach(button => button.classList.remove('active'));


            document.getElementById(cityName).classList.add('active');

            event.target.classList.add('active');

        }


        // Accordion

        function toggleAccordion(element) {

            const content = element.nextElementSibling;

            const isActive = element.classList.contains('active');


            // Fecha todos

            document.querySelectorAll('.accordion-header').forEach(header => {

                header.classList.remove('active');

                header.nextElementSibling.classList.remove('active');

            });


            // Abre o clicado se não estava ativo

            if (!isActive) {

                element.classList.add('active');

                content.classList.add('active');

            }

        }

        // Calculadora de custos

        function calcularCusto() {

            const numCasais = parseInt(document.getElementById('numCasais').value) || 1;

            const barco = parseFloat(document.getElementById('barco').value) || 0;

            const restaurantes = parseFloat(document.getElementById('restaurantes').value) || 1500;


            // Custos fixos por casal (média)

            const aereo = 11800; // R$ 5.900 x 2 pessoas

            const hospedagem = 19200; // média

            const carro = 2960; // média

            const combustivel = 448; // média

            const estacionamento = 1152; // média

            const beachClubs = 3200; // média

            const ferries = 1056; // média

            const diversos = 2560; // média


            // Barco dividido pelos casais

            const barcoPorCasal = (barco / numCasais) * 6.4; // converter para reais


            // Restaurantes em reais

            const restaurantesReais = restaurantes * 6.4;


            const totalPorCasal = aereo + hospedagem + carro + combustivel + 

                                  estacionamento + restaurantesReais + beachClubs + 

                                  ferries + diversos + barcoPorCasal;


            const totalGrupo = totalPorCasal * numCasais;


            const resultado = document.getElementById('resultado');

            resultado.style.display = 'block';

            resultado.innerHTML = `

                <p>💰 Total por casal: <span style="color: var(--secondary);">R$ ${totalPorCasal.toLocaleString('pt-BR', {minimumFractionDigits: 2})}</span></p>

                <p style="margin-top: 1rem;">👥 Total do grupo (${numCasais} casais): <span style="color: var(--primary);">R$ ${totalGrupo.toLocaleString('pt-BR', {minimumFractionDigits: 2})}</span></p>

            `;

        }
